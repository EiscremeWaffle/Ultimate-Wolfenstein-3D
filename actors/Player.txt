actor BJPlayerU : BJPlayer
{
	health 100
	radius 22
	player.movebob 0
	player.viewheight 32
	player.displayname "BJ Blazkowicz"
	player.startitem "Clip", 8
	player.startitem "Pistol"
	player.startitem "Knife"
	player.weaponslot 1, "Knife", "Knife2"
	player.weaponslot 2, "Pistol", "BluePistol"
	player.weaponslot 3, "MachineGun", "BlueAK47"
	player.weaponslot 4, "GatlingGun", "BlueGatlingGun"
	player.weaponslot 5, "FlameThrower"
	player.weaponslot 6, "RocketLauncher"
	player.damagescreencolor "FF 00 00"
	states
	{
		Spawn:
			PLAY A -1
			stop
		See:
			PLAY B 6.5
			PLAY C 4
			PLAY D 6.5
			PLAY E 4
			loop
		Missile:
			PLAY FG 10
			goto See
		Melee:
			PLAY H 10 bright
			goto See
		Pain:
			PLAY I 5 A_JumpIf(health & 1, 1)
			goto See
			PLAY J 5
			goto See
		Death:
			PLAY K 7.5 A_Fall
			PLAY L 7.5
			PLAY M 7.5
			PLAY N -1
			stop
	}
}
