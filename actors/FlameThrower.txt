actor FlameThrower : WolfWeapon 211
{
	inventory.pickupsound "weapon/flame/pickup"
	weapon.ammotype1 "GasTank"
	weapon.ammogive1 99
	states
	{
		Spawn:
			FTHR A -1
			stop
		Ready:
			FLMG A 1 A_WeaponReady
			loop
		Fire:
			FLMG B 3
		Hold:
			FLMG C 3 bright A_FireCustomMissile("Flame")
			FLMG D 3 bright A_FireCustomMissile("Flame")
			FLMG B 3 A_ReFire
			goto Ready
	}
}
